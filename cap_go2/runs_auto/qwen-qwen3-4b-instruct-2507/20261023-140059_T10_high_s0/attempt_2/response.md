```python
# 壁にぶつからないように、安全な経路を選び、赤い箱まで確実に進む
# 壁の中心 (2.0, 0.0)、大きさ 0.3×1.6（y方向幅1.6m）→ y方向範囲: [-0.8, 0.8]
# 赤い箱は (4.0, 0.0) にあり、壁の前に進む必要がある。

# 1. 初期位置 (0.0, 0.0) から、壁に近づきすぎないよう、y方向に0.4mずらして前に進む
#    これにより、壁のy方向範囲（±0.8）内に留まり、ぶつからないようにする

move_sideways(0.4)  # y方向に0.4m右にずらす（ロボットが壁の上にいない）

# 2. その後、x方向に進むが、壁のx=2.0までに到達する前に、y方向を調整して壁に近づかないようにする
#    壁のx方向は2.0、ロボットは(0.4, 0.0) → x方向に1.6m進む必要がある

# 3. 壁にぶつからないように、x方向に進む（1.6m）
move_forward(1.6)

# 4. 現在の位置を確認（壁の前に到達したか）
x, y = get_position()
print(f"壁の前到達: ({x}, {y})")

# 5. 現在のy座標が壁のy範囲内か確認（-0.8 <= y <= 0.8）
if y < -0.8 or y > 0.8:
    # y方向が壁の範囲外なら、y方向に調整
    move_sideways(-0.4)  # y方向に0.4m戻す

# 6. 赤い箱 (4.0, 0.0) に向かって進む
#    今の位置からx方向に(4.0 - x)だけ進む
distance_to_goal = 4.0 - x
move_forward(distance_to_goal)

# 7. 到着確認（赤い箱の中心 (4.0, 0.0) まで0.3m以内）
target_x, target_y = 4.0, 0.0
final_x, final_y = get_position()
distance = math.hypot(target_x - final_x, target_y - final_y)

print(f"赤い箱までの距離: {distance:.2f}m")

# 8. 距離が0.3m以内なら停止、そうでなければ少し近づく
if distance > 0.3:
    move_forward(0.5)

# 9. 最終的に停止
stop(1.0)
```
