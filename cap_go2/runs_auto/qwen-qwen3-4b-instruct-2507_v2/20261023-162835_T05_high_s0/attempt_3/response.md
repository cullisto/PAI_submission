```python
# 1辺1メートルの正方形を正確に描くため、4つの辺をそれぞれ正確な方向で移動し、
# すべての移動と旋回を90度の方向に制御（0° → 90° → 180° → 270° → 360°=0°）。
# 各移動は現在の向きに沿って1.0m前進、旋回はその場で90度右回り（-90度）に変化。

# 初期位置: (0.0, 0.0), 初期向き: 0 deg (x軸正方向)

# 正方形の1辺の長さ
side_length = 1.0

# 1. x方向に前進（+x）
move_forward(side_length)
print("前進1.0m (x方向)")

# 2. 右に90度旋回（-90度） → 現在の向きは +y方向
turn(-90)
print("右に90度旋回")

# 3. y方向に前進（+y）
move_forward(side_length)
print("前進1.0m (y方向)")

# 4. 右に90度旋回（-90度） → 現在の向きは -x方向
turn(-90)
print("右に90度旋回")

# 5. x方向に前進（-x）
move_forward(-side_length)
print("前進1.0m (-x方向)")

# 6. 右に90度旋回（-90度） → 現在の向きは -y方向
turn(-90)
print("右に90度旋回")

# 7. y方向に前進（-y）
move_forward(-side_length)
print("前進1.0m (-y方向)")

# 8. 右に90度旋回（-90度） → 現在の向きは +x方向（元の向きに戻る）
turn(-90)
print("右に90度旋回")

# 9. 最後に停止（元の位置に戻っていれば、位置が(0,0)に近づく）
stop(1.0)
```
